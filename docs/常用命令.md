# 常用命令

# 1. 构建 LLVM Pass
cd /src/code/passes/build
cmake .. -DLLVM_DIR=/usr/lib/llvm-11/cmake
make -j4

# 2. 构建运行时库
cd /src/code/runtime
clang-11 -c -O2 -fPIC array_state_runtime.c -o array_state_runtime.o
ar rcs libarray_state.a array_state_runtime.o

# 3. 运行静态分析
opt-11 -load ./passes/build/libarray_analysis_pass.so \
    -array-analysis target.ll -o /dev/null 2> analysis.log

# 4. 编译插桩版本
clang-11 -S -emit-llvm -g -O0 target.c -o target.ll
opt-11 -load ./passes/build/libarray_instrument_pass.so \
    -array-instrument target.ll -S -o target_instr.ll
clang-11 target_instr.ll ./runtime/libarray_state.a \
    -o target_instr -lm

# 5. 启动 AFLGO 模糊测试
AFL_NO_UI=1 AFL_SKIP_CPUFREQ=1 AFL_ARRAY_DIVERSITY=1 \
    /opt/aflgo/afl-2.57b/afl-fuzz \
    -i seeds -o output -- ./target_instr @@

# 6. 查看反馈文件
cat /tmp/array_feedback.txt

# 7. docker
docker run -it --rm -v "$(pwd):/workspace" \\
    -w /workspace ubuntu:18.04 bash
